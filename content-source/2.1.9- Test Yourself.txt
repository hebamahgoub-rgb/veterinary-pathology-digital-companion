    let userAnswers = new Array(10).fill(null);

    function checkAnswer(qIdx, optionIdx) {
      if (userAnswers[qIdx] !== null) return; // Prevent re-answering

      userAnswers[qIdx] = optionIdx;
      const qCard = document.querySelector(`.q-card[data-q="${qIdx}"]`);
      const buttons = qCard.querySelectorAll('.option-btn');
      const expBox = document.getElementById(`exp-${qIdx}`);
      const correctIdx = quizData[qIdx].correct;

      // Disable buttons and mark styles
      buttons.forEach((btn, idx) => {
        btn.disabled = true;
        if (idx === correctIdx) {
          btn.classList.add('correct');
        } else if (idx === optionIdx) {
          btn.classList.add('incorrect');
        }
      });

      // Show dynamic explanation based on answer correctness
      if (optionIdx === correctIdx) {
        expBox.textContent = `Correct! ${quizData[qIdx].exp}`;
        expBox.className = 'q-explanation show-correct';
      } else {
        const correctLetter = String.fromCharCode(65 + correctIdx);
        expBox.textContent = `Incorrect. Option ${correctLetter} is the correct answer. ${quizData[qIdx].exp}`;
        expBox.className = 'q-explanation show-incorrect';
      }

      updateScore();
    }

    function updateScore() {
      let score = 0;
      let answeredCount = 0;

      userAnswers.forEach((ans, idx) => {
        if (ans !== null) {
          answeredCount++;
          if (ans === quizData[idx].correct) score++;
        }
      });

      const percent = Math.round((score / 10) * 100);
      const scoreDisplay = document.getElementById('score-display');
      const scoreFeedback = document.getElementById('score-feedback');

      scoreDisplay.textContent = `${score} / 10 (${percent}%)`;

      if (answeredCount === 10) {
        if (percent >= 90) {
          scoreFeedback.textContent = "Outstanding! Mastered all concepts in Disturbance in Cell Metabolism.";
        } else if (percent >= 70) {
          scoreFeedback.textContent = "Good job! Solid understanding. Review missed topics in the module.";
        } else {
          scoreFeedback.textContent = "Keep reviewing! Re-read the sub-pages and try the quiz again.";
        }
      } else {
        scoreFeedback.textContent = `Progress: ${answeredCount} of 10 questions answered.`;
      }
    }

    function resetQuiz() {
      userAnswers = new Array(10).fill(null);
      const qCards = document.querySelectorAll('.q-card');

      qCards.forEach((card, qIdx) => {
        const buttons = card.querySelectorAll('.option-btn');
        const expBox = document.getElementById(`exp-${qIdx}`);

        buttons.forEach(btn => {
          btn.disabled = false;
          btn.classList.remove('correct', 'incorrect');
        });

        expBox.className = 'q-explanation';
        expBox.textContent = '';
      });

      document.getElementById('score-display').textContent = '0 / 10 (0%)';
      document.getElementById('score-feedback').textContent = 'Answer the questions below to test your understanding.';
    }
